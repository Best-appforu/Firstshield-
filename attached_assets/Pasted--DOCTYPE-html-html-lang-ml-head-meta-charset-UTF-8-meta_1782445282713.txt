        // 1. Mobile App Safety Tabs logic
        function switchTab(evt, tabId) {
            const card = evt.currentTarget.closest('.card');
            const tabButtons = card.querySelectorAll('.tab-btn');
            const tabContents = card.querySelectorAll('.tab-content');

            tabButtons.forEach(btn => btn.classList.remove('active'));
            tabContents.forEach(content => content.classList.remove('active'));

            evt.currentTarget.classList.add('active');
            card.querySelector(`#${tabId}`).classList.add('active');
        }

        // 2. Privacy Score Calculator logic
        function updateScore() {
            const checkboxes = document.querySelectorAll('.perm-check');
            let deduction = 0;
            
            checkboxes.forEach(cb => {
                if (cb.checked) {
                    deduction += parseInt(cb.value);
                }
            });

            const finalScore = 100 - deduction;
            const scoreNum = document.getElementById('scoreNum');
            const scoreStatus = document.getElementById('scoreStatus');
            const scoreBox = document.getElementById('scoreBox');

            scoreNum.innerText = finalScore + "%";

            // Dynamically change visual states & Malayalam messages based on score
            if (finalScore === 100) {
                scoreStatus.innerText = "സുരക്ഷിതമാണ്! ആശങ്കപ്പെടേണ്ടതില്ല.";
                scoreBox.style.backgroundColor = "var(--pastel-blue)";
            } else if (finalScore === 75) {
                scoreStatus.innerText = "ശ്രദ്ധിക്കുക: പ്രൈവസി ചെറുതായി ബാധിക്കപ്പെട്ടു.";
                scoreBox.style.backgroundColor = "var(--pastel-yellow)";
            } else if (finalScore === 50) {
                scoreStatus.innerText = "അപകടസാധ്യത! കൂടുതൽ അനുമതികൾ നൽകിയിരിക്കുന്നു.";
                scoreBox.style.backgroundColor = "var(--pastel-pink)";
            } else if (finalScore <= 25) {
                scoreStatus.innerText = "തീർത്തും സുരക്ഷിതമല്ല! വിവരങ്ങൾ ചോരാൻ സാധ്യതയേറി.";
                scoreBox.style.backgroundColor = "#FFD6D6";
            }
        }

        }
